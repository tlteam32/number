วิธีการเชื่อมต่อ Google Sheets

  1. สร้างสเปรดชีตใหม่ใน Google Sheets
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้วคัดลอกโค้ดด้านล่างนี้ไปวาง:
function doGet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const result = [];
  
  // Skip header if exists
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] !== "") {
      result.push({
        number: String(data[i][0]),
        name: String(data[i][1] || ''),
        updatedAt: String(data[i][2] || '')
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    const data = JSON.parse(e.postData.contents);
    
    if (data.action === 'saveAll') {
      // Clear sheet and rewrite header + rows
      sheet.clearContents();
      sheet.appendRow(['Number', 'Name', 'UpdatedAt']);
      
      data.items.forEach(item => {
        sheet.appendRow([item.number, item.name, item.updatedAt]);
      });
      
      return ContentService.createTextOutput(JSON.stringify({status: 'success'}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
